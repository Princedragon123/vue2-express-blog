关注功能前后端设计文档
一、核心问题分析
当前问题根源：数据状态不一致
用户详情页、关注列表页、数据库三者状态不同步
前端缓存与后端实际数据脱节
关注操作缺乏事务性和状态同步机制
二、设计原则
单一数据源：关注状态以用户详情接口为准
原子操作：关注/取关操作要保证数据一致性
实时同步：操作后立即同步所有相关状态
幂等设计：重复操作不会产生副作用

3. 后端实现关键逻辑
Python（Flask）示例：
from flask import request, jsonify
from flask_jwt_extended import jwt_required, get_jwt_identity
from datetime import datetime
import redis

class FollowService:
    def __init__(self):
        self.redis_client = redis.Redis(host='localhost', port=6379, db=0)
    
    def follow_user(self, current_user_id, target_user_id):
        """关注用户 - 核心逻辑"""
        
        # 1. 验证不能关注自己
        if current_user_id == target_user_id:
            return {"error": "不能关注自己"}, 400
        
        # 2. 检查是否已关注（幂等性检查）
        if self.is_following(current_user_id, target_user_id):
            return {"error": "已经关注该用户"}, 400
        
        try:
            # 3. 开启数据库事务
            with db.session.begin():
                # 4. 插入关注关系
                follow = UserFollow(
                    follower_id=current_user_id,
                    following_id=target_user_id,
                    created_at=datetime.utcnow()
                )
                db.session.add(follow)
                
                # 5. 更新用户统计（原子操作）
                User.query.filter_by(id=current_user_id).update({
                    'following_count': User.following_count + 1
                })
                User.query.filter_by(id=target_user_id).update({
                    'followers_count': User.followers_count + 1
                })
            
            # 6. 清除相关缓存
            self._clear_follow_cache(current_user_id, target_user_id)
            
            # 7. 返回最新状态
            return {
                "is_following": True,
                "followers_count": self.get_followers_count(target_user_id),
                "following_count": self.get_following_count(current_user_id)
            }, 200
            
        except Exception as e:
            db.session.rollback()
            return {"error": "关注失败"}, 500
    
    def is_following(self, follower_id, following_id):
        """检查关注关系"""
        # 先查缓存
        cache_key = f"follow:{follower_id}:{following_id}"
        cached = self.redis_client.get(cache_key)
        if cached is not None:
            return cached.decode() == '1'
        
        # 查数据库
        exists = UserFollow.query.filter_by(
            follower_id=follower_id,
            following_id=following_id
        ).first() is not None
        
        # 写入缓存（有效期5分钟）
        self.redis_client.setex(cache_key, 300, '1' if exists else '0')
        return exists
    
    def _clear_follow_cache(self, follower_id, target_user_id):
        """清除关注相关缓存"""
        # 清除关注状态缓存
        self.redis_client.delete(f"follow:{follower_id}:{target_user_id}")
        # 清除关注列表缓存
        self.redis_client.delete(f"following_list:{follower_id}")
        self.redis_client.delete(f"followers_list:{target_user_id}")
        # 清除用户详情缓存
        self.redis_client.delete(f"user:{follower_id}")
        self.redis_client.delete(f"user:{target_user_id}")

# API路由
@app.route('/api/v1/users/<target_user_id>/follow', methods=['POST'])
@jwt_required()
def follow_user(target_user_id):
    current_user_id = get_jwt_identity()
    follow_service = FollowService()
    
    result, status_code = follow_service.follow_user(current_user_id, target_user_id)
    return jsonify(result), status_code

// store/modules/user.js
export default {
  state: {
    // 用户关注状态映射 { userId: isFollowing }
    followStatus: {},
    // 关注列表数据
    followingList: [],
    followersList: []
  },
  
  mutations: {
    SET_FOLLOW_STATUS(state, { userId, isFollowing }) {
      // Vue.set 确保响应式
      Vue.set(state.followStatus, userId, isFollowing)
    },
    
    SET_FOLLOWING_LIST(state, list) {
      state.followingList = list
      // 同步更新关注状态
      list.forEach(user => {
        Vue.set(state.followStatus, user.id, true)
      })
    }
  },
  
  actions: {
    async toggleFollow({ commit, state }, targetUserId) {
      const currentStatus = state.followStatus[targetUserId]
      const newStatus = !currentStatus
      
      try {
        // 乐观更新：先更新UI
        commit('SET_FOLLOW_STATUS', { userId: targetUserId, isFollowing: newStatus })
        
        // 发送请求
        const method = newStatus ? 'POST' : 'DELETE'
        const response = await api.followUser(targetUserId, method)
        
        if (response.code === 200) {
          // 成功：更新计数等
          return response.data
        } else {
          // 失败：回滚状态
          commit('SET_FOLLOW_STATUS', { 
            userId: targetUserId, 
            isFollowing: currentStatus 
          })
          throw new Error(response.message)
        }
      } catch (error) {
        // 处理特定错误
        if (error.message.includes('已经关注')) {
          // 已经是关注状态，确保前端状态正确
          commit('SET_FOLLOW_STATUS', { 
            userId: targetUserId, 
            isFollowing: true 
          })
        }
        throw error
      }
    },
    
    async refreshFollowingLists({ commit, dispatch }, userId) {
      // 并行获取关注和粉丝列表
      const [followingRes, followersRes] = await Promise.all([
        api.getUserFollowing(userId),
        api.getUserFollowers(userId)
      ])
      
      commit('SET_FOLLOWING_LIST', followingRes.data.list)
      commit('SET_FOLLOWERS_LIST', followersRes.data.list)
      
      // 触发组件更新
      dispatch('notifyDataUpdated')
    }
  }
}
2. 关注按钮组件
<template>
  <button 
    :class="['follow-btn', { 'following': isFollowing }]"
    :disabled="loading || isSelf"
    @click="handleFollow"
  >
    <span v-if="loading">处理中...</span>
    <span v-else-if="isSelf">自己</span>
    <span v-else>{{ isFollowing ? '已关注' : '关注' }}</span>
  </button>
</template>

<script>
export default {
  name: 'FollowButton',
  props: {
    userId: {
      type: String,
      required: true
    },
    // 初始状态（从用户详情接口获取）
    initialFollowing: {
      type: Boolean,
      default: false
    }
  },
  
  data() {
    return {
      loading: false,
      // 本地状态，优先使用全局store的状态
      localIsFollowing: this.initialFollowing
    }
  },
  
  computed: {
    // 优先使用全局状态
    isFollowing() {
      return this.$store.state.user.followStatus[this.userId] ?? this.localIsFollowing
    },
    
    currentUser() {
      return this.$store.state.user.currentUser
    },
    
    isSelf() {
      return this.currentUser && this.currentUser.id === this.userId
    }
  },
  
  methods: {
    async handleFollow() {
      if (this.isSelf || this.loading) return
      
      this.loading = true
      
      try {
        // 调用全局action
        await this.$store.dispatch('user/toggleFollow', this.userId)
        
        // 成功提示
        this.$message.success(this.isFollowing ? '关注成功' : '已取消关注')
        
        // 广播状态更新事件，让其他组件刷新
        this.$emit('follow-changed', this.isFollowing)
        this.$bus.$emit('follow-status-updated', {
          userId: this.userId,
          isFollowing: this.isFollowing
        })
        
      } catch (error) {
        // 错误已在action中处理，这里只做UI反馈
        this.$message.error(error.message || '操作失败')
      } finally {
        this.loading = false
      }
    }
  },
  
  created() {
    // 监听全局关注状态更新事件
    this.$bus.$on('follow-status-updated', (data) => {
      if (data.userId === this.userId) {
        this.localIsFollowing = data.isFollowing
      }
    })
  },
  
  beforeDestroy() {
    this.$bus.$off('follow-status-updated')
  }
}
</script>

        </div>
        <follow-button

      </div>
    </div>
  </div>

<script>
export default {
  name: 'FollowingList',
  
  data() {
    return {
      list: [],
      loading: false,
      refreshing: false,
      pagination: {
        page: 1,
        hasMore: true
      }
    }
  },
  
  computed: {
    userId() {
      return this.$route.params.userId || this.$store.state.user.currentUser.id
    }
  },
  
  async created() {
    await this.loadList()
    
    // 监听关注状态变化
    this.$bus.$on('follow-status-updated', this.handleFollowUpdate)
  },
  
  beforeDestroy() {
    this.$bus.$off('follow-status-updated', this.handleFollowUpdate)
  },
  
  methods: {
    async loadList(forceRefresh = false) {
      if (this.loading) return
      
      this.loading = true
      
      try {
        // 添加时间戳防止缓存
        const timestamp = Date.now()
        const params = {
          page: this.pagination.page,
          limit: 20,
          _t: forceRefresh ? timestamp : undefined
        }
        
        const response = await this.$api.get(`/users/${this.userId}/following`, { params })
        
        if (forceRefresh || this.pagination.page === 1) {
          this.list = response.data.list
        } else {
          this.list = [...this.list, ...response.data.list]
        }
        
        this.pagination.hasMore = response.data.pagination.has_more
        
        // 更新全局状态
        this.list.forEach(user => {
          this.$store.commit('user/SET_FOLLOW_STATUS', {
            userId: user.id,
            isFollowing: user.is_following
          })
        })
        
      } catch (error) {
        console.error('加载关注列表失败:', error)
        this.$message.error('加载失败')
      } finally {
        this.loading = false
        this.refreshing = false
      }
    },
    
    async refreshList() {
      this.refreshing = true
      this.pagination.page = 1
      await this.loadList(true)
    },
    
    onFollowChanged(isFollowing) {
      // 如果取消关注，从列表中移除
      if (!isFollowing) {
        // 可以根据实际情况决定是否立即移除
        // 或者等待下次刷新
        this.$message.info('已取消关注，列表将在下次刷新时更新')
      }
    },
    
    handleFollowUpdate(data) {
      // 更新列表中的用户状态
      const index = this.list.findIndex(user => user.id === data.userId)
      if (index !== -1) {
        this.list[index].is_following = data.isFollowing
        // 强制更新视图
        this.$set(this.list, index, { ...this.list[index] })
      }
    }
  }
}
</script>

// utils/sync.js
class FollowStatusSync {
  constructor() {
    this.syncInterval = null
    this.syncingUsers = new Set()
  }
  
  // 启动同步（在用户登录后调用）
  start() {
    // 每30秒同步一次关键用户的关注状态
    this.syncInterval = setInterval(() => {
      this.syncImportantUsers()
    }, 30000)
  }
  
  // 同步重要用户（当前页面查看的用户）
  async syncImportantUsers() {
    const usersToSync = Array.from(this.syncingUsers)
    if (usersToSync.length === 0) return
    
    try {
      const requests = usersToSync.map(userId => 
        this.$api.getUserInfo(userId)
      )
      
      const responses = await Promise.all(requests)
      
      responses.forEach(response => {
        const user = response.data
        this.$store.commit('user/SET_FOLLOW_STATUS', {
          userId: user.id,
          isFollowing: user.is_following
        })
      })
    } catch (error) {
      console.warn('同步关注状态失败:', error)
    }
  }
  
  // 添加需要同步的用户
  addUser(userId) {
    this.syncingUsers.add(userId)
  }
  
  // 移除同步用户
  removeUser(userId) {
    this.syncingUsers.delete(userId)
  }
  
  // 停止同步
  stop() {
    if (this.syncInterval) {
      clearInterval(this.syncInterval)
      this.syncInterval = null
    }
  }
}

七、常见问题解决方案
问题1：关注后列表不立即更新
解决方案：操作成功后发送全局事件，列表组件监听事件并刷新
问题2：多个标签页状态不同步
解决方案：使用 localStorage 或 BroadcastChannel 跨标签页同步
问题3：统计数字不一致
解决方案：使用数据库事务，同时更新关系表和计数表
问题4：缓存雪崩
解决方案：缓存设置随机过期时间，数据库查询使用互斥锁
