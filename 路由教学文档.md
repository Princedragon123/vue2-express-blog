# 🎨 前端路由（Vue Router）

## 1️⃣ 什么是前端路由？

前端路由负责**页面之间的切换**，不需要重新加载整个页面。

### 🎯 核心概念

| 概念 | 说明 |
|------|------|
| **路由（Router）** | 管理页面跳转的"指挥中心" |
| **路由配置** | 定义 URL 对应哪个组件 |
| **路由守卫** | 页面跳转前的检查（如登录验证） |
| **`this.$router.push()`** | 编程式跳转页面 |

---

## 2️⃣ 最简版前端路由

```javascript
// main.js（前端入口）
import Vue from 'vue';
import Router from 'vue-router';
import App from './App.vue';

// 使用路由插件
Vue.use(Router);

// 导入组件
import Login from './components/Login.vue';
import Blog from './components/Blog.vue';

// 创建路由实例
const router = new Router({
  mode: 'history',
  routes: [
    {
      path: '/login',      // 当 URL 是 /login 时
      name: 'login',
      component: Login      // 显示 Login 组件
    },
    {
      path: '/blog',       // 当 URL 是 /blog 时
      name: 'blog',
      component: Blog       // 显示 Blog 组件
    },
    {
      path: '/',
      redirect: '/blog'     // 访问根路径时重定向到 /blog
    }
  ]
});

// 创建 Vue 应用并挂载路由
const app = new Vue({
  router,
  render: h => h(App)
}).mount('#app');
```

---

## 3️⃣ 路由守卫（页面跳转前的检查）

```javascript
// 在 router/index.js 中
router.beforeEach((to, from, next) => {
  // to: 要跳转到哪里
  // from: 从哪里来
  // next: 放行函数
  
  console.log('从', from.path, '跳转到', to.path);
  
  // 公开路由（不需要登录就能访问）
  const publicRoutes = ['/login', '/register', '/blog'];
  
  if (!auth.isLoggedIn() && !publicRoutes.includes(to.path)) {
    // 没登录，且访问的不是公开路由 → 跳转到登录页
    next('/login');
  } else {
    // 其他情况 → 正常放行
    next();
  }
});
```

## 4️⃣ 如何在组件中使用路由

### 方式1：编程式跳转

```vue
<template>
  <button @click="goToBlog">去博客首页</button>
</template>

<script>
export default {
  methods: {
    goToBlog() {
      // 跳转到 /blog
      this.$router.push('/blog');
    }
  }
}
</script>
```

### 方式2：获取路由参数

```vue
<script>
export default {
  mounted() {
    // 获取 URL 中的参数（比如 /profile/123）
    const userId = this.$route.params.userId;
    console.log('用户ID:', userId);
  }
}
</script>
```

# 🔧 后端路由（Express）

## 1️⃣ 什么是后端路由？

后端路由负责**接收前端的 API 请求**，处理业务逻辑，返回数据。

### 🎯 核心概念

| 概念 | 说明 |
|------|------|
| **app（Express 实例）** | 后端主心骨，监听端口 |
| **router（子路由）** | 把不同功能的路由分组 |
| **controller（控制器）** | 具体的业务逻辑代码 |
| **HTTP 方法** | GET（获取）、POST（创建）、PUT（更新）、DELETE（删除） |

---

## 2️⃣ 最简版后端（单文件版）

```javascript
// server.js（后端入口，所有代码写在一个文件里）
const express = require('express');

// 1. 创建 app 实例（主心骨）
const app = express();

// 2. 配置中间件（解析 JSON 请求体）
app.use(express.json());

// 3. 定义路由（app 直接处理请求）

// GET 请求：获取数据
app.get('/api/hello', (req, res) => {
  res.send('你好，世界！');
});

// POST 请求：登录
app.post('/api/auth/login', (req, res) => {
  const { username, password } = req.body;
  
  // 简单验证
  if (username === 'admin' && password === '123456') {
    res.json({ 
      success: true, 
      message: '登录成功',
      token: 'fake-jwt-token'
    });
  } else {
    res.status(400).json({ 
      success: false, 
      message: '用户名或密码错误' 
    });
  }
});

// 4. 启动服务器
app.listen(3001, () => {
  console.log('服务器启动在 http://localhost:3001');
});
```

## 3️⃣ 为什么要拆分路由？

### ❌ 不拆分的问题

```javascript
// server.js 会变得超级长！
const app = express();

app.get('/api/hello', ...);
app.post('/api/auth/login', ...);
app.post('/api/auth/register', ...);
app.get('/api/users', ...);
app.get('/api/users/:id', ...);
app.get('/api/blogs', ...);
app.post('/api/blogs', ...);
// ... 几十上百个路由！

app.listen(3001, ...);
```

---

### ✅ 拆分后的结构

```
bloglogin/
├── src/
│   ├── server.js              # 后端入口（主心骨）
│   ├── routes/
│   │   ├── index.js           # 主路由（组织所有子路由）
│   │   ├── authRoutes.js      # 认证路由
│   │   ├── userRoutes.js      # 用户路由
│   │   └── blogRoutes.js      # 博客路由
│   ├── controllers/
│   │   ├── authController.js  # 认证控制器
│   │   ├── userController.js  # 用户控制器
│   │   └── blogController.js  # 博客控制器
```

## 4️⃣ 拆分后的代码

### 文件1：`server.js`（后端入口）

```javascript
// src/server.js
const express = require('express');
const app = express();

// 导入主路由文件
const routes = require('./routes/index.js');

// 【关键】把所有路由挂载到 /api 前缀下
// 意思：所有 /api 开头的请求，都交给 routes 去处理
app.use('/api', routes);

app.listen(3001, () => {
  console.log('服务器启动在 http://localhost:3001');
});
```

---

### 文件2：`routes/index.js`（主路由）

```javascript
// src/routes/index.js
const express = require('express');
// 创建一个 router 实例（就像一个小的 app）
const router = express.Router();

// 导入各个子路由
const authRoutes = require('./authRoutes');
const blogRoutes = require('./blogRoutes');

// 【关键】把子路由挂载到不同的前缀下
router.use('/auth', authRoutes);   // /api/auth 开头的请求
router.use('/blogs', blogRoutes);  // /api/blogs 开头的请求

module.exports = router;
```

---

### 文件3：`routes/authRoutes.js`（认证子路由）

```javascript
// src/routes/authRoutes.js
const express = require('express');
const router = express.Router();

// 导入控制器
const authController = require('../controllers/authController');

// 【关键】定义具体的路由，交给控制器处理
router.post('/register', authController.register);
router.post('/login', authController.login);

module.exports = router;
```

---

### 文件4：`controllers/authController.js`（控制器）

```javascript
// src/controllers/authController.js
exports.login = async (req, res) => {
  const { email, password } = req.body;
  
  try {
    // 1. 查找用户
    const user = await User.findOne({ 
      $or: [{ email }, { username: email }] 
    });
    
    if (!user) {
      return res.status(400).json({ 
        success: false, 
        message: '邮箱或密码错误' 
      });
    }
    
    // 2. 验证密码
    const isMatch = await user.comparePassword(password);
    
    if (!isMatch) {
      return res.status(400).json({ 
        success: false, 
        message: '邮箱或密码错误' 
      });
    }
    
    // 3. 生成 JWT token
    const token = jwt.generateToken(user._id, user.role);
    
    // 4. 返回响应
    res.json({ 
      success: true, 
      message: '登录成功',
      token,
      user: {
        id: user._id,
        username: user.username,
        email: user.email
      }
    });
  } catch (error) {
    res.status(500).json({ 
      success: false, 
      message: '服务器内部错误' 
    });
  }
};
```

---

## 5️⃣ `app.use()` vs `router.use()` 的区别

| 方法 | 作用 | 在哪里用 |
|------|------|----------|
| `app.use()` | 把中间件/路由挂载到 app（主实例）上 | `server.js` |
| `router.use()` | 把子路由挂载到 router（子实例）上 | `routes/index.js`、子路由文件 |

# 🔄 完整调用流程

## 🎯 举例：用户登录

### 请求路径
```
POST http://localhost:3001/api/auth/login
```

### 调用链

```
┌─────────────────────────────────────────────────────┐
│ 1. 用户在前端点击"登录"按钮                          │
│    Login.vue 组件调用 this.$http.auth.login()        │
└─────────────────────┬───────────────────────────────┘
                      ↓
┌─────────────────────────────────────────────────────┐
│ 2. 前端 api.js 发送 POST 请求                        │
│    请求地址：/api/auth/login                          │
└─────────────────────┬───────────────────────────────┘
                      ↓
┌─────────────────────────────────────────────────────┐
│ 3. 后端 server.js 接收请求                           │
│    匹配到 app.use('/api', routes)                    │
└─────────────────────┬───────────────────────────────┘
                      ↓
┌─────────────────────────────────────────────────────┐
│ 4. routes/index.js 接收 /api 开头的请求             │
│    匹配到 router.use('/auth', authRoutes)            │
└─────────────────────┬───────────────────────────────┘
                      ↓
┌─────────────────────────────────────────────────────┐
│ 5. authRoutes.js 接收 /api/auth 开头的请求          │
│    匹配到 router.post('/login', authController.login)│
└─────────────────────┬───────────────────────────────┘
                      ↓
┌─────────────────────────────────────────────────────┐
│ 6. authController.login 执行登录逻辑                  │
│    - 查找用户                                         │
│    - 验证密码                                         │
│    - 生成 JWT token                                   │
└─────────────────────┬───────────────────────────────┘
                      ↓
┌─────────────────────────────────────────────────────┐
│ 7. 返回响应给前端                                     │
│    res.json({ success: true, token: ... })           │
└─────────────────────┬───────────────────────────────┘
                      ↓
┌─────────────────────────────────────────────────────┐
│ 8. 前端 api.js 响应拦截器处理响应                    │
│    保存 token 到 localStorage                         │
└─────────────────────┬───────────────────────────────┘
                      ↓
┌─────────────────────────────────────────────────────┐
│ 9. 前端调用 this.$router.push('/blog') 跳转页面    │
└─────────────────────────────────────────────────────┘
```

# 📝 总结

| 部分 | 作用 | 主心骨 |
|------|------|--------|
| **前端** | 页面切换、权限控制 | `main.js` → `router/index.js` |
| **后端** | 接收 API 请求、处理业务逻辑 | `server.js` → `routes/index.js` |

现在明白了吗？整个架构就像两棵树，前端和后端各自独立，通过 API 连接在一起！🐱✨
